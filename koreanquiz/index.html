// 데이터를 가볍게 쓰기 위해 배열로 압축한 것을 객체로 파싱하는 함수 const quizData = { 1: [], 2: [], 3: [] }; for (let stage in rawData) { quizData[stage] = rawData[stage].map(item => ({ word: item[0], answer: item[1], options: [item[1], item[2], item[3]], exp: item[4], meaning: item[5] || "" })); } let currentStage = 1; let currentQuestionIndex = 0; let score = 0; let activeQuestions = []; const menuScreen = document.getElementById('menu-screen'); const quizScreen = document.getElementById('quiz-screen'); const resultScreen = document.getElementById('result-screen'); const optionsContainer = document.getElementById('options-container'); const feedbackEl = document.getElementById('feedback'); const expEl = document.getElementById('explanation'); const nextBtn = document.getElementById('next-btn'); function showMenu() { menuScreen.classList.remove('hidden'); quizScreen.classList.add('hidden'); resultScreen.classList.add('hidden'); } function startStage(stage) { currentStage = stage; currentQuestionIndex = 0; score = 0; // 핵심: 100개의 데이터 중 배열을 무작위로 섞은 뒤(Shuffle), 설정한 갯수(10개)만 잘라냄 const shuffledPool = [...quizData[stage]].sort(() => Math.random() - 0.5); activeQuestions = shuffledPool.slice(0, QUESTIONS_PER_ROUND); document.getElementById('stage-title').innerText = `Stage ${stage}`; menuScreen.classList.add('hidden'); quizScreen.classList.remove('hidden'); loadQuestion(); } function loadQuestion() { feedbackEl.innerText = ""; expEl.classList.add('hidden'); nextBtn.classList.add('hidden'); optionsContainer.innerHTML = ""; const q = activeQuestions[currentQuestionIndex]; document.getElementById('progress-text').innerText = `Question ${currentQuestionIndex + 1} / ${activeQuestions.length}`; document.getElementById('question-word').innerText = q.word; const meaningEl = document.getElementById('word-meaning'); if (q.meaning) { meaningEl.innerText = `Meaning: ${q.meaning}`; meaningEl.classList.remove('hidden'); } else { meaningEl.classList.add('hidden'); } // 보기 섞기 const shuffledOptions = [...q.options].sort(() => Math.random() - 0.5); shuffledOptions.forEach(opt => { const btn = document.createElement('button'); btn.className = 'option-btn'; btn.innerText = opt; btn.onclick = () => checkAnswer(btn, opt, q); optionsContainer.appendChild(btn); }); } function checkAnswer(clickedBtn, selectedWord, q) { const allBtns = optionsContainer.querySelectorAll('.option-btn'); allBtns.forEach(btn => { btn.disabled = true; if(btn.innerText === q.answer) btn.classList.add('correct'); }); if (selectedWord === q.answer) { feedbackEl.innerText = "✨ Correct!"; feedbackEl.className = "feedback-area feedback-correct"; score++; } else { clickedBtn.classList.add('wrong'); feedbackEl.innerText = `❌ Wrong! (Answer: ${q.answer})`; feedbackEl.className = "feedback-area feedback-wrong"; // 틀렸을 때 설명 박스 띄우기 expEl.innerText = `💡 Tip: ${q.exp}`; expEl.classList.remove('hidden'); } nextBtn.classList.remove('hidden'); } function nextQuestion() { currentQuestionIndex++; if (currentQuestionIndex < activeQuestions.length) loadQuestion(); else showResults(); } function showResults() { quizScreen.classList.add('hidden'); resultScreen.classList.remove('hidden'); document.getElementById('final-score').innerText = `${score} / ${activeQuestions.length}`; }
